
import java.util.ArrayList;
import java.util.Scanner;

class Student {
    String name, rollNumber, course;
    double totalFee, paidFee, balance;

    Student(String name, String rollNumber, String course,
            double totalFee, double paidFee) {
        this.name = name;
        this.rollNumber = rollNumber;
        this.course = course;
        this.totalFee = totalFee;
        this.paidFee = paidFee;
        this.balance = totalFee - paidFee;
    }
}

public class FeeManagementSystem {
    static ArrayList<Student> students = new ArrayList<>();
    static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        int choice;

        do {
            System.out.println("\n===== Student Fee Management System =====");
            System.out.println("1. Add Student Fee Payment");
            System.out.println("2. View All Student Records");
            System.out.println("3. Search Student Record");
            System.out.println("4. Generate Receipt");
            System.out.println("5. Exit");
            System.out.print("Enter your choice: ");

            choice = sc.nextInt();
            sc.nextLine();

            switch (choice) {
                case 1:
                    addStudent();
                    break;
                case 2:
                    viewAllStudents();
                    break;
                case 3:
                    searchStudent();
                    break;
                case 4:
                    generateReceipt();
                    break;
                case 5:
                    System.out.println("Exiting system. Thank you!");
                    break;
                default:
                    System.out.println("Invalid choice! Try again.");
            }
        } while (choice != 5);
    }

    static void addStudent() {
        System.out.print("Enter Student Name: ");
        String name = sc.nextLine();

        System.out.print("Enter Roll Number: ");
        String roll = sc.nextLine();

        System.out.print("Enter Course Name: ");
        String course = sc.nextLine();

        System.out.print("Enter Total Fee: ");
        double totalFee = sc.nextDouble();

        System.out.print("Enter Paid Fee: ");
        double paidFee = sc.nextDouble();
        sc.nextLine();

        double balance = totalFee - paidFee;
        System.out.println("Calculated Balance = " + balance);

        Student student = new Student(
            name, roll, course, totalFee, paidFee
        );

        students.add(student);
        System.out.println("Student fee payment recorded successfully!");
    }

    static void viewAllStudents() {
        if (students.isEmpty()) {
            System.out.println("No records found.");
            return;
        }

        System.out.println("\nAll Student Records:");

        System.out.printf("%-15s %-10s %-20s %-10s %-10s %-10s%n",
            "Name", "Roll No", "Course", "Total Fee", "Paid", "Balance");

        for (Student s : students) {
            System.out.printf("%-15s %-10s %-20s %-10.2f %-10.2f %-10.2f%n",
                s.name, s.rollNumber, s.course,
                s.totalFee, s.paidFee, s.balance);
        }
    }

    static void searchStudent() {
        System.out.print("Enter Roll Number to search: ");
        String roll = sc.nextLine();

        boolean found = false;

        for (Student s : students) {
            if (s.rollNumber.equalsIgnoreCase(roll)) {
                System.out.println("Student Found!");

                System.out.printf(
                    "Name: %s%nRoll Number: %s%nCourse: %s%nTotal Fee: %.2f%nPaid Fee: %.2f%nBalance: %.2f%n",
                    s.name, s.rollNumber, s.course,
                    s.totalFee, s.paidFee, s.balance
                );

                found = true;
                break;
            }
        }

        if (!found) {
            System.out.println("Student record not found!");
        }
    }

    static void generateReceipt() {
        System.out.print("Enter Roll Number for receipt: ");
        String roll = sc.nextLine();

        boolean found = false;

        for (Student s : students) {
            if (s.rollNumber.equalsIgnoreCase(roll)) {
                System.out.println("\n------ FEE RECEIPT ------");
                System.out.println("Student Name : " + s.name);
                System.out.println("Roll Number  : " + s.rollNumber);
                System.out.println("Course       : " + s.course);
                System.out.printf("Total Fee    : %.2f%n", s.totalFee);
                System.out.printf("Amount Paid  : %.2f%n", s.paidFee);
                System.out.printf("Remaining Fee: %.2f%n", s.balance);
                System.out.println("-------------------------");
                System.out.println("Thank You!");

                found = true;
                break;
            }
        }

        if (!found) {
            System.out.println("Student record not found!");
        }
    }
}
